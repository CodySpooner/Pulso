def test_month_old_queued_claim_still_blocks_retry_and_replay(
    session: Session, settings: Settings, s3: MagicMock, monkeypatch: pytest.MonkeyPatch
) -> None:
    real_execute = import_service.execute_run
    monkeypatch.setattr(
        import_service, "execute_run", lambda s, run_id, *a, **k: s.get_one(CollectorRun, run_id)
    )
    body = template_csv("google_trends")
    first = commit_import(session, "google_trends", body, settings=settings)
    assert first.status == "queued"
    first.queued_at = datetime.now(UTC) - timedelta(days=30)  # far beyond any stale_after
    session.commit()
    monkeypatch.setattr(import_service, "execute_run", real_execute)
    retry = commit_import(session, "google_trends", body, settings=settings)
    payload_id = session.scalars(select(RawPayload.id)).one()
    replay = replay_import(session, payload_id, settings=settings)
    assert (retry.status, retry.params.get("reason")) == ("skipped", "in_progress")
    assert retry.params["original_run_id"] == first.id
    assert (replay.status, replay.params.get("reason")) == ("skipped", "in_progress")
